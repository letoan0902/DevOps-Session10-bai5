$ cat docker-compose.yml
# Docker Compose v2 - cụm Backend Web API + PostgreSQL (đóng gói từ bài 4)
# Giá trị ${...} được Compose nội suy từ tệp .env cùng thư mục.

services:
  # Dịch vụ cơ sở dữ liệu
  database:
    image: postgres:15-alpine
    restart: unless-stopped
    environment:
      POSTGRES_DB: ${DB_NAME}
      POSTGRES_USER: ${DB_USER}
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    volumes:
      # Named volume giữ dữ liệu khi container bị xóa
      - pgdata:/var/lib/postgresql/data
    networks:
      - backend-net
    # Kiểm tra DB đã sẵn sàng nhận kết nối chưa
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${DB_USER} -d ${DB_NAME}"]
      interval: 5s
      timeout: 5s
      retries: 10

  # Dịch vụ Backend Web API (build từ ./backend)
  backend:
    build: ./backend
    restart: unless-stopped
    ports:
      # Cổng host lấy từ .env, cổng container cố định 8080
      - "${PORT}:8080"
    environment:
      DB_HOST: database
      DB_PORT: ${DB_PORT}
      DB_NAME: ${DB_NAME}
      DB_USER: ${DB_USER}
      DB_PASSWORD: ${DB_PASSWORD}
    # Dạng dài: chờ database healthy rồi mới khởi động backend
    depends_on:
      database:
        condition: service_healthy
    networks:
      - backend-net

# Volume và network khai báo ở cấp root, Compose tự tạo khi up
volumes:
  pgdata:

networks:
  backend-net:
